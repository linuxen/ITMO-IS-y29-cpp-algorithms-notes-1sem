Class 1) Инкапсуляция

Это парадигма с++ как есть функциональное так есть и ООП

cpp
#include <iostream>

class Human {
public:

метод - функция принадлежащая классу

Метод с квалификатором const не может изменять обычные поля объекта. Поля с mutable составляют исключение.

Если метод можно сделать const значит он обязательно должен быть const иначе будет ошибка в лабе

cpp
#include <iostream>

class Human {
public:
    int age = 18;
    std::string name;

    void Print() { // Метод
        std::cout << name << '\t' << age << '\n';
    }
};

int main() {
    Human Name1;
    Name1.age = 18;
    Name1.name = "Alex";

    Name1.Print();
}

Геттеры и Сеттеры

cpp
#include <iostream>

class Human {
private:
    int age = 18;
    std::string name = "Alex";
public:
    int Get(){
        return age;
    }

    void Set(int set_age) {
        age = set_age;
    }

    void Print() {
        std::cout << age << '\n';
    }
};

int main() {
    Human Name;

    // GETTER
    int result = Name.Get();
    std::cout << result << '\n';

    // SETTER
    int new_age = 19;
    Name.Set(new_age);
    Name.Print();
}

Конструктор

Чтобы создать объект класса без мусора в начале и чтобы избежать бесконечных getterov and setter

Поля — это переменные, объявленные внутри класса, которые хранят данные объекта

Конструктор - вызывается по имени класса

cpp
#include <iostream>

class Human {
private:
    int age;
    std::string name;

public:
	// Конструктор
    Human (int value_age, std::string ind_name){
        age = value_age;
        name = ind_name;
    }

    int Get(){
        return age;
    }

    void Set(int set_age) {
        age = set_age;
    }

    void Print() {
        std::cout << age << '\n';
    }

};

int main() {
    Human Name(18, "Alex"); // Конструктор
    int new_age = 19;

    // GETTER
    int result = Name.Get();
    std::cout << result << '\n';

    // SETTER
    Name.Set(new_age);
    Name.Print();
}

конструктор копирования

реализуется неявно если не задано ничего

Запускается при копировании класса в функцию

cpp
#include <iostream>

class MyClass {
public:
    int* data = new int[size];
    MyClass(int size){ // Constructor
        std::cout << "Constructor called" << '\n';
        this->size = size;

        for (int i = 0; i < size; i++) {
            data[i] = i;
        }
    }

    ~MyClass(){ // Destructor
        std::cout << "called Destructor" << '\n';
        delete[] data;
    }

private:
    int size;
};

int main() {
    MyClass Hello(10);
    MyClass Hello2(Hello);
}
text
Constructor called
Copy called
called Destructor
called Destructor

При поверхностном копировании владеющего указателя оба объекта обращаются к одному ресурсу. При уничтожении обоих объектов один и тот же ресурс освобождается повторно. Копирующий конструктор должен создать отдельную копию ресурса либо копирование нужно запретить.

cpp
#include <iostream>

class MyClass {
public:
    int* data;
    int copy_size;

    MyClass(int size){ // Constructor
        std::cout << "Constructor called" << '\n';
        this->data = new int[size];
        this->copy_size = size;

        for (int i = 0; i < size; i++) {
            data[i] = i;
        }
    }

    MyClass(const MyClass& other){ //Copy called
        std::cout << "Copy called" << '\n';
        this->data = new int[other.copy_size];
        for (int i = 0; i < other.copy_size; i++){
            other.data[i] = i;
        }
    }

    ~MyClass(){ // Destructor
        std::cout << "called Destructor" << '\n';
        delete[] data;
    }
};

int main() {
    MyClass Hello(10);
    MyClass Hello2(Hello);
}
text
Constructor called
Copy called
called Destructor
called Destructor

В этом примере копирование создаёт самостоятельную копию ресурса.

Это по умолчанию

cpp
MyClass(const MyClass& other){
        this->data = other.data;
    }

Деструктор

Когда выполнение выходит из области видимости в main, автоматически вызываются деструкторы локальных объектов.

Деструктор не принимает параметры

cpp
#include <iostream>

class MyClass {
public:
    MyClass(){ // Constructor
        std::cout << "called constructor" << '\n';
    }
    ~MyClass(){ // Destructor
        std::cout << "called Destructor" << '\n';
    }
};

int main() {
    MyClass a;
}
text
called constructor
called Destructor

cpp
#include <iostream>

class MyClass {

public:
    MyClass(int new_age, std::string new_name){ // Constructor
        age = new_age;
        name = new_name;
    }

    ~MyClass(){ // Destructor
        std::cout << "called Destructor" << '\n';
    }

    void Get() {
        std::cout << this->age << '\n';
    }

    void Set(int age) {
        this->age = age; // this->age обозначает поле, age — параметр конструктора
    }

private:
    int age;
    std::string name;
};

int main() {
    MyClass Hello(18, "Alex");
    Hello.Get();

    Hello.Set(20);
    Hello.Get();
}

explicit

explicit - означает явное указание компилятору, что конструктор или оператор преобразования не должен использоваться для неявного (автоматического) преобразования типов, что предотвращает ошибки и неожиданное поведение
